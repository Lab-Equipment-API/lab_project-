  "equipment_count": 3,
  "created_at": "...",

{ "id": 5, "name": "Robotics & Automation", "created_at": "...", "updated_at": "..." }


--- DELETE /categories/:id (no equipment) ---

==============================================================
  TASK 7 — EDGE CASES (All 10 Scenarios)
==============================================================

---------------------------------------------------------
EDGE CASE 1: Create equipment with non-existent category_id
---------------------------------------------------------
curl -X POST http://localhost:3000/equipment \
  -H "Content-Type: application/json" \
  -d '{"equipment": {"name": "Mystery Device", "serial_number": "MYS-001", "status": "available", "category_id": 999}}'

Expected 422:
{ "error": "Category not found" }


---------------------------------------------------------
EDGE CASE 2: Create equipment with duplicate serial number
---------------------------------------------------------
(Assuming ARD-001 already exists in the database)
curl -X POST http://localhost:3000/equipment \
  -H "Content-Type: application/json" \
  -d '{"equipment": {"name": "Another Arduino", "serial_number": "ARD-001", "status": "available", "category_id": 4}}'

Expected 422:
{ "errors": ["Serial number has already been taken"] }


---------------------------------------------------------
EDGE CASE 3: Create equipment with invalid status "broken"
---------------------------------------------------------
curl -X POST http://localhost:3000/equipment \
  -H "Content-Type: application/json" \
  -d '{"equipment": {"name": "Broken Thing", "serial_number": "BRK-001", "status": "broken", "category_id": 4}}'

Expected 422:
{ "errors": ["Status is not included in the list"] }


---------------------------------------------------------
EDGE CASE 4: Create category with duplicate name
---------------------------------------------------------
(Assuming "Computing" already exists)
curl -X POST http://localhost:3000/categories \
  -H "Content-Type: application/json" \
  -d '{"category": {"name": "Computing"}}'

Expected 422:
{ "errors": ["Name has already been taken"] }


---------------------------------------------------------
EDGE CASE 5: Create maintenance record with non-existent equipment_id
---------------------------------------------------------
curl -X POST http://localhost:3000/maintenance_records \
  -H "Content-Type: application/json" \
  -d '{"maintenance_record": {"description": "Test service", "performed_at": "2024-03-01T10:00:00Z", "equipment_id": 999}}'

Expected 422:
{ "error": "Equipment not found" }


---------------------------------------------------------
EDGE CASE 6: DELETE /categories/:id when equipment belongs to it
---------------------------------------------------------
(Assuming category id=1 "Computing" has equipment items)
curl -X DELETE http://localhost:3000/categories/1

Expected 409:
{ "error": "Cannot delete category. 3 equipment items still belong to it." }

(The number reflects however many equipment items belong to category 1 in your seed data)


---------------------------------------------------------
EDGE CASE 7: GET /categories/:id that does not exist
---------------------------------------------------------
curl -X GET http://localhost:3000/categories/999

Expected 404:
{ "error": "Category not found" }


---------------------------------------------------------
EDGE CASE 8: GET /equipment/:id that does not exist
---------------------------------------------------------
curl -X GET http://localhost:3000/equipment/999

Expected 404:
{ "error": "Equipment not found" }


---------------------------------------------------------
EDGE CASE 9: PATCH /categories/:id that does not exist
---------------------------------------------------------
curl -X PATCH http://localhost:3000/categories/999 \
  -H "Content-Type: application/json" \
  -d '{"category": {"name": "Ghost Category"}}'

Expected 404:
{ "error": "Category not found" }


---------------------------------------------------------
EDGE CASE 10: POST /maintenance_records with future performed_at
---------------------------------------------------------
curl -X POST http://localhost:3000/maintenance_records \
  -H "Content-Type: application/json" \
  -d '{"maintenance_record": {"description": "Scheduled maintenance", "performed_at": "2099-01-01T10:00:00Z", "equipment_id": 1}}'

Expected 422:
{ "errors": ["Performed at cannot be in the future"] }


==============================================================
  ADDITIONAL BUSINESS RULE TESTS (Task 6)
==============================================================

--- Rule 1: Invalid serial number format ---
curl -X POST http://localhost:3000/equipment \
  -H "Content-Type: application/json" \
  -d '{"equipment": {"name": "Bad Serial", "serial_number": "lap-001", "status": "available", "category_id": 1}}'

Expected 422:
{ "errors": ["Serial number must match format XXX-NNN (e.g. LAP-001)"] }

curl -X POST http://localhost:3000/equipment \
  -H "Content-Type: application/json" \
  -d '{"equipment": {"name": "Bad Serial 2", "serial_number": "LAP01", "status": "available", "category_id": 1}}'

Expected 422:
{ "errors": ["Serial number must match format XXX-NNN (e.g. LAP-001)"] }

curl -X POST http://localhost:3000/equipment \
  -H "Content-Type: application/json" \
  -d '{"equipment": {"name": "Bad Serial 3", "serial_number": "LP-001", "status": "available", "category_id": 1}}'

Expected 422:
{ "errors": ["Serial number must match format XXX-NNN (e.g. LAP-001)"] }


--- Rule 3: Category name too short ---
curl -X POST http://localhost:3000/categories \
  -H "Content-Type: application/json" \
  -d '{"category": {"name": "AB"}}'

Expected 422:
{ "errors": ["Name is too short (minimum is 3 characters)"] }

Expected 422:
{ "errors": ["Name must contain at least one letter"] }

curl -X POST http://localhost:3000/equipment \
  -H "Content-Type: application/json" \
  -d '{"equipment": {"name": "AB", "serial_number": "TST-002", "status": "available", "category_id": 1}}'

Expected 422:
{ "errors": ["Name is too short (minimum is 3 characters)"] }
